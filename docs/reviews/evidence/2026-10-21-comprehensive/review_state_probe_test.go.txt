func TestReviewCloseDuringOnBarStopsAdvance(t *testing.T) {
	base := Generate(10, 1)
	all := map[string][]Candle{"ALPHA": base}
	c, _ := replayChart(t, []string{"ALPHA"}, all, LayoutStack)
	r, err := StartReplay(c, all)
	if err != nil {
		t.Fatal(err)
	}
	callbacks := 0
	r.OnBar(func(ReplayBar) { callbacks++; r.Close() })
	err = r.Step(5)
	t.Logf("after callback Close: Step err=%v closed=%v cursor=%d bars=%d callbacks=%d", err, r.closed, r.Cursor(), c.Bars(), callbacks)
	if r.Cursor() != 1 || c.Bars() != 1 || callbacks != 1 {
		t.Fatal("the detached replay continued to mutate the chart")
	}
}

func TestReviewStartReplayDuringOnBarPreservesNewCursor(t *testing.T) {
	base := Generate(10, 1)
	all := map[string][]Candle{"ALPHA": base}
	c, _ := replayChart(t, []string{"ALPHA"}, all, LayoutStack)
	old, err := StartReplay(c, all)
	if err != nil {
		t.Fatal(err)
	}
	var replacement *Replay
	old.OnBar(func(ReplayBar) {
		old.OnBar(nil)
		replacement, err = StartReplay(c, all)
		if err != nil {
			t.Fatal(err)
		}
	})
	err = old.Step(4)
	if replacement == nil {
		t.Fatal("callback did not run")
	}
	t.Logf("old Step err=%v old closed=%v old cursor=%d new cursor=%d chart bars=%d attachedNew=%v", err, old.closed, old.Cursor(), replacement.Cursor(), c.Bars(), c.replay == replacement)
	if c.Bars() != 0 || replacement.Cursor() != 0 {
		t.Fatal("old replay exposed bars that the attached replacement did not reveal")
	}
}

func TestReviewPauseDuringOnBarStopsTick(t *testing.T) {
	base := Generate(10, 1)
	all := map[string][]Candle{"ALPHA": base}
	c, _ := replayChart(t, []string{"ALPHA"}, all, LayoutStack)
	r, err := StartReplay(c, all)
	if err != nil {
		t.Fatal(err)
	}
	now := time.Date(2024, 3, 1, 0, 0, 0, 0, time.UTC)
	r.SetClock(func() time.Time { return now })
	r.OnBar(func(ReplayBar) { r.Pause() })
	r.Play(10)
	now = now.Add(500 * time.Millisecond)
	r.Tick()
	t.Logf("paused=%v cursor=%d bars=%d", !r.Playing(), r.Cursor(), c.Bars())
	if r.Cursor() != 1 {
		t.Fatal("Tick continued after callback paused playback")
	}
}

func TestReviewCoarseReplayHidesFutureMarkers(t *testing.T) {
	start := time.Date(2024, 3, 1, 8, 0, 0, 0, time.UTC)
	base := sampleWalk(8, 1, start, time.Hour)
	all := map[string][]Candle{"ALPHA": base}
	c, _ := replayChart(t, []string{"ALPHA"}, all, LayoutStack)
	r, err := StartReplay(c, all)
	if err != nil {
		t.Fatal(err)
	}
	if err := c.SetMarkers("ALPHA", []Marker{{Time: base[2].Time, Side: SideBuy, Label: "HIDDEN_TRADE"}}); err != nil {
		t.Fatal(err)
	}
	if err := r.Step(2); err != nil {
		t.Fatal(err)
	}
	t.Logf("before 4h: cursor=%d marker bar=%d", r.Cursor(), c.panes[0].marks[0].bar)
	if err := c.SetPeriod(Period{Span: 4, Unit: UnitHour}); err != nil {
		t.Fatal(err)
	}
	c.crossIndex, c.crossPane = 0, 0
	readout := c.Readout("ALPHA")
	t.Logf("after 4h: cursor=%d newestRevealed=%s markerTime=%s marker bar=%d readout=%q", r.Cursor(), base[1].Time, base[2].Time, c.panes[0].marks[0].bar, readout)
	if c.panes[0].marks[0].bar >= 0 || strings.Contains(readout, "HIDDEN_TRADE") {
		t.Fatal("future trade is visible before its base bar is revealed")
	}
}

func TestReviewCloseThenSamePeriodResubscribes(t *testing.T) {
	p := Period{Span: 1, Unit: UnitHour}
	all := Generate(2, 1)
	src := &StubLiveSource{}
	if err := src.SetBars("ALPHA", p, all[:1]); err != nil {
		t.Fatal(err)
	}
	c := NewChart(ChartOptions{Source: src, Symbols: []string{"ALPHA"}, Period: p, Indicators: []string{}})
	if err := c.Close(); err != nil {
		t.Fatal(err)
	}
	if err := c.SetPeriod(c.Period()); err != nil {
		t.Fatal(err)
	}
	src.Push("ALPHA", p, all[1])
	t.Logf("bars=%d liveSubs=%d loadError=%v", c.Bars(), len(c.liveSubs), c.LoadError())
	if c.Bars() != 2 {
		t.Fatal("same-period reopen did not resubscribe")
	}
}
